const ProfileScreen = ({ profiles, setProfiles, onSelectProfile }) => {
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [activeTab, setActiveTab] = useState('xtream');
  const [isConnecting, setIsConnecting] = useState(false);
  const [progress, setProgress] = useState(0);
  const [statusText, setStatusText] = useState('Connexion au serveur...');

  // Form State
  const [pName, setPName] = useState('');
  const [url, setUrl] = useState('');
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');

  const handleCreateProfile = () => {
    setIsConnecting(true);
    setProgress(10);
    setStatusText('Connexion aux serveurs IPTV...');

    const interval = setInterval(() => {
      setProgress((prev) => {
        if (prev >= 100) {
          clearInterval(interval);
          setTimeout(() => {
            const newProfile = {
              id: Date.now().toString(),
              name: pName || (activeTab === 'xtream' ? 'Compte IPTV' : 'Playlist M3U'),
              type: activeTab === 'xtream' ? 'Xtream Codes' : 'Lien M3U',
              url: url,
            };
            setProfiles([...profiles, newProfile]);
            setIsConnecting(false);
            setIsModalOpen(false);
            // Reset fields
            setPName(''); setUrl(''); setUsername(''); setPassword('');
          }, 300);
          return 100;
        }
        if (prev === 40) setStatusText('Téléchargement du guide EPG...');
        if (prev === 75) setStatusText('Synchronisation des chaînes & films...');
        return prev + 15;
      });
    }, 180);
  };

  const handleDeleteProfile = (e, id) => {
    e.stopPropagation();
    setProfiles(profiles.filter((p) => p.id !== id));
  };

  return (
    <div className="w-screen h-screen glass-bg-animated flex flex-col items-center justify-center relative p-8">
      {/* Top Brand Banner */}
      <div className="absolute top-10 left-12 flex items-center gap-4">
        <div className="w-14 h-14 rounded-2xl bg-gradient-to-tr from-indigo-600 to-purple-500 flex items-center justify-center text-white font-extrabold text-2xl shadow-lg shadow-indigo-500/20">
          V
        </div>
        <div>
          <h1 className="text-3xl font-extrabold text-slate-800 tracking-tight">VisionTV</h1>
          <p className="text-xs font-semibold text-indigo-600 tracking-widest uppercase">WebOS IPTV Player</p>
        </div>
      </div>

      {/* Screen Content */}
      {profiles.length === 0 ? (
        /* Empty State */
        <div className="flex flex-col items-center text-center max-w-lg">
          <div className="w-32 h-32 rounded-3xl glass-card flex items-center justify-center mb-8 shadow-2xl text-indigo-600">
            <Icons.Tv className="w-16 h-16" />
          </div>
          <h2 className="text-4xl font-extrabold text-slate-800 mb-4 tracking-tight">Bienvenue sur VisionTV</h2>
          <p className="text-slate-600 text-lg mb-10 leading-relaxed">
            Aucun profil n'est configuré. Ajoutez votre premier compte Xtream ou votre liste M3U pour démarrer.
          </p>
          <button
            onClick={() => setIsModalOpen(true)}
            className="tv-focusable flex items-center gap-4 bg-indigo-600 text-white px-10 py-5 rounded-2xl font-bold text-xl shadow-xl shadow-indigo-600/30 hover:bg-indigo-500 transition-all"
          >
            <Icons.Plus className="w-7 h-7" />
            Ajouter un profil
          </button>
        </div>
      ) : (
        /* Active Profiles Grid */
        <div className="w-full max-w-5xl flex flex-col items-center">
          <h2 className="text-4xl font-extrabold text-slate-800 mb-3 tracking-tight">Sélectionnez votre profil</h2>
          <p className="text-slate-500 text-lg mb-12">Choisissez un compte pour accéder au direct et à la VOD</p>

          <div className="flex flex-wrap justify-center gap-8 w-full">
            {profiles.map((p) => (
              <div key={p.id} className="relative group">
                <button
                  onClick={() => onSelectProfile(p)}
                  className="tv-focusable w-64 h-64 glass-card rounded-3xl p-6 flex flex-col items-center justify-center text-center cursor-pointer border-2 border-white/80 hover:bg-white/80 transition-all"
                >
                  <div className="w-20 h-20 rounded-2xl bg-indigo-100 text-indigo-600 flex items-center justify-center mb-5 shadow-inner">
                    {p.type.includes('Xtream') ? <Icons.User className="w-10 h-10" /> : <Icons.Link className="w-10 h-10" />}
                  </div>
                  <h3 className="text-2xl font-bold text-slate-800 truncate w-full mb-1">{p.name}</h3>
                  <span className="text-xs font-bold text-indigo-500 uppercase tracking-wider bg-indigo-50 px-3 py-1 rounded-full border border-indigo-100">
                    {p.type}
                  </span>
                </button>

                {/* Delete Profile Button */}
                <button
                  onClick={(e) => handleDeleteProfile(e, p.id)}
                  className="tv-focusable absolute top-4 right-4 w-10 h-10 rounded-xl bg-red-500/10 text-red-600 flex items-center justify-center hover:bg-red-500 hover:text-white transition-all border border-red-200"
                  title="Supprimer ce profil"
                >
                  <Icons.Trash className="w-5 h-5" />
                </button>
              </div>
            ))}
          </div>

          {/* Bottom Right Floating Add Button */}

      {/* Add Profile Modal */}
      {isModalOpen && (

                <button

                >
                  <Icons.X className="w-6 h-6" />
                </button>
              )}
            </div>

              <div className="py-12 flex flex-col items-center">
                <div className="w-full bg-slate-200 rounded-full h-4 mb-6 overflow-hidden">
                  <div
                    className="bg-gradient-to-r from-indigo-500 to-purple-500 h-full transition-all duration-300 ease-out"
                    style={{ width: `${progress}%` }}
                  />
                </div>
                <p className="text-xl font-bold text-indigo-600 animate-pulse">{statusText}</p>
                <span className="text-sm text-slate-500 font-semibold mt-2">{progress}%</span>

                {/* Tabs */}
                <div className="flex gap-2 p-1.5 bg-slate-200/60 rounded-2xl mb-8">

                          className="w-1/2 bg-white/80 border border-slate-200 rounded-2xl px-5 py-4 text-slate-800 placeholder-slate-400 focus:outline-none focus:ring-4 focus:ring-indigo-300 tv-focusable text-lg shadow-sm"
                        />
                      </div>

                    <input
